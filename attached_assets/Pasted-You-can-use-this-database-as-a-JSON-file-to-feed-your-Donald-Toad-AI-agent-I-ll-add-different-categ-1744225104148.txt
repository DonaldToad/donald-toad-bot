import random
import json
from telegram import Update
from telegram.ext import Updater, CommandHandler, MessageHandler, Filters

# Load the crypto knowledge from the JSON file
def load_crypto_knowledge():
    with open("crypto_knowledge.json", "r") as file:
        data = json.load(file)
    return data

# Global variable to hold the crypto knowledge
crypto_knowledge = load_crypto_knowledge()

# Handle the /start command
def start(update: Update, context):
    update.message.reply_text("Welcome to Donald Toad's Crypto Knowledge! Type any crypto question, ribbit! 🐸")

# Handle crypto-related questions
def handle_crypto_questions(update: Update, context):
    user_input = update.message.text.lower()

    if "blockchain" in user_input:
        response = crypto_knowledge["crypto_terminology"]["blockchain"]["definition"]
    elif "ethereum" in user_input:
        response = crypto_knowledge["crypto_terminology"]["ethereum"]["definition"]
    elif "bitcoin" in user_input:
        response = crypto_knowledge["top_cryptos"]["bitcoin"]["fun_fact"]
    elif "cardano" in user_input:
        response = crypto_knowledge["top_cryptos"]["cardano"]["fun_fact"]
    elif "inflation" in user_input:
        response = crypto_knowledge["crypto_humor"]["inflation"]["joke"]
    elif "vitalik" in user_input:
        response = crypto_knowledge["founders_and_personalities"]["vitalik_buterin"]["fun_fact"]
    else:
        response = "Sorry, I don't know that one. Try asking about blockchain, Bitcoin, Ethereum, or something crypto-related! Croak!"

    # Send the response
    update.message.reply_text(response)

# Main function to handle the bot
def main():
    updater = Updater("YOUR_BOT_TOKEN", use_context=True)
    updater.dispatcher.add_handler(CommandHandler("start", start))
    updater.dispatcher.add_handler(MessageHandler(Filters.text, handle_crypto_questions))

    updater.start_polling()
    updater.idle()

if __name__ == "__main__":
    main()
